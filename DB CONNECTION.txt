function getIceData() {
    global $connection;

    // Fetch all ice cream datas
    $query = "SELECT * FROM icecreams";
    
    $result = pg_query($connection, $query);


    if ($result === false) {
        echo json_encode(["message" => "Error: " . pg_last_error($connection)]);
        return;
    }
    //fetched data is assigned to $icecreams
    $icecreams = pg_fetch_all($result);
    //echo json_encode($icecreams);exit;

    if ($icecreams === false) {
        echo json_encode(["status" => "failure", "data" => []]);
        return;
    }

    // Fetch associated images and thumbnails for each ice cream entry
    foreach ($icecreams as &$icecream) {
        $ice_id = $icecream['ice_id'];

        // Fetch images for the current ice cream
        $query = "SELECT * FROM icecreams_image WHERE ice_id = $1";
        $result = pg_query_params($connection, $query, [$ice_id]);
        if ($result !== false) {
            $images = pg_fetch_all($result);
           // echo json_encode($images);exit;
            $icecream['image'] = $images !== false ? $images : [];
        } else {
            $icecream['image'] = [];
        }

        // Fetch thumbnails for the current ice cream
        $query = "SELECT * FROM thumbnail WHERE ice_id = $1";
        $result = pg_query_params($connection, $query, [$ice_id]);
        if ($result !== false) {
            $thumbnails = pg_fetch_all($result);
            $icecream['thumbnail'] = $thumbnails !== false ? $thumbnails : [];
        } else {
            $icecream['thumbnail'] = [];
        }
    }

    // Return the result as a JSON response
    echo json_encode(["status" => "success", "data" => $icecreams]);
}

class Validator {
    private $errors = [];

    public function validate($data, $rules = [], $labels = []) {
        foreach ($rules as $field => $fieldRules) {
            $value = $data[$field] ?? '';
            foreach ($fieldRules as $rule) {
                $this->applyRule($field, $value, $rule, $labels[$field] ?? ucfirst($field));
            }
        }
        return $this->errors;
    }

    private function applyRule($field, $value, $rule, $label) {
        list($ruleName, $ruleVal) = is_array($rule) ? $rule : [$rule, true];

        if ($ruleName === 'required' && empty($value)) {
            $this->addError("$label is required");
        } elseif ($ruleName === 'minLen' && strlen($value) < $ruleVal) {
            $this->addError("$label should be a minimum of $ruleVal characters");
        } elseif ($ruleName === 'maxLen' && strlen($value) > $ruleVal) {
            $this->addError("$label should be a maximum of $ruleVal characters");
        }
    }

    private function addError($message) {
        $this->errors[] = $message;
    }
}
